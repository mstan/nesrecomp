cmake_minimum_required(VERSION 3.20)
project(nesref LANGUAGES CXX)
find_package(SDL2 CONFIG QUIET)
if(NOT TARGET SDL2::SDL2)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
endif()
add_executable(nesref frontend.cpp)
target_compile_features(nesref PRIVATE cxx_std_11)
target_include_directories(nesref PRIVATE ../../runner/src)
if(TARGET SDL2::SDL2)
    target_link_libraries(nesref PRIVATE SDL2::SDL2)
else()
    target_link_libraries(nesref PRIVATE PkgConfig::SDL2)
endif()
